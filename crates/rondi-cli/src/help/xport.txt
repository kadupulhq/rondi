Usage: rrdtool [options] command command_options
* xport - generate XML dump from one or several RRD

	rrdtool xport [-s|--start seconds] [-e|--end seconds]
		[-m|--maxrows rows]
		[--step seconds]
		[-t|--showtime]
		[--enumds] [--json]
		[-d|--daemon address]
		[DEF:vname=rrd:ds-name:CF]
		[CDEF:vname=rpn-expression]
		[XPORT:vname:legend]
